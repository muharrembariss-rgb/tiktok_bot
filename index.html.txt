    <script>
        let simdikiSoru = 1;
        let toplamSoru = 7;
        let puan = 0;
        let kullaniciCevaplari = [];

        function testiBaslat() {
            document.getElementById('login-section').style.display = 'none';
            document.getElementById('quiz-section').style.display = 'block';
            document.getElementById('q1').style.display = 'block';
        }

        function cevapVer(buton, dogruMu, mesaj) {
            kullaniciCevaplari.push({
                soruNo: simdikiSoru,
                secim: buton.innerText,
                dogruMu: dogruMu
            });

            let soruKutusu = buton.parentElement;
            let butonlar = soruKutusu.getElementsByTagName('button');
            for(let b of butonlar) {
                b.disabled = true;
            }

            if(dogruMu) {
                buton.classList.add('correct');
                puan++;
            } else {
                buton.classList.add('wrong');
            }

            let feedbackDiv = soruKutusu.getElementsByClassName('feedback-msg')[0];
            feedbackDiv.style.display = 'block';
            feedbackDiv.className = dogruMu ? 'feedback-msg feedback-success' : 'feedback-msg feedback-error';
            feedbackDiv.innerHTML = mesaj;

            if (simdikiSoru < toplamSoru) {
                document.getElementById('next-btn').style.display = 'block';
            } else {
                document.getElementById('score').style.display = 'block';
                document.getElementById('puan-yazisi').innerText = puan;
                let msgBox = document.getElementById('result-message');
                msgBox.style.display = 'block';
                
                if (puan === toplamSoru) {
                    msgBox.className = 'success';
                    msgBox.innerHTML = 'Mükemmelsin Arifenur! Testi fulledin, seni çok seviyorum!';
                } else {
                    msgBox.className = 'warning';
                    msgBox.innerHTML = 'Olsun sevgilim, puanın: ' + puan + ' / 7. Seni çok seviyorum!';
                }

                fetch('/sonucu-kaydet', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ puan: puan, cevaplar: kullaniciCevaplari })
                });
            }
        }

        function sonrakiSoru() {
            document.getElementById('q' + simdikiSoru).style.display = 'none';
            document.getElementById('next-btn').style.display = 'none';

            simdikiSoru++;
            document.getElementById('q' + simdikiSoru).style.display = 'block';
            document.getElementById('progress-subtitle').innerText = 'Soru ' + simdikiSoru + ' / ' + toplamSoru;
        }
    </script>
